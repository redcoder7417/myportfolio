    List<BookingResult> results = new List<BookingResult>();

async def analyze_candidate(resume_text, job_desc):
    vector_store = PineconeStore()
    docs = await vector_store.similarity_search(query=job_desc, k=3)
    context = "\n".join([d.text for d in docs])
    prompt = f"Compare resume with job:\n{context}\nResume: {resume_text}"
    return await llm.generate(prompt)
